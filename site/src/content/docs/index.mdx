import { Card, CardGrid, LinkButton } from "@astrojs/starlight/components"

```ts import.meta.vitest
import { Replica, ReplicatedText } from "@jpowersdev/effect-delta"
import { Effect } from "effect"

const program = Effect.gen(function*() {
  const laptop = yield* ReplicatedText.make({
    replica: Replica.makeId("laptop")
  })
  const phone = yield* ReplicatedText.make({
    replica: Replica.makeId("phone")
  })

  // Offline, each edits their own copy
  yield* ReplicatedText.insert(laptop, 0, "Hello")
  yield* ReplicatedText.insert(phone, 0, "world")

  // Back online, both show the same text
  yield* Replica.exchange(laptop, phone)
  const text = yield* ReplicatedText.get(phone)
  yield* Effect.log(text)
})

await Effect.runPromise(program)
```

    yield* Replica.exchange(laptop, phone)

  <Card title="Sync over any transport" icon="random">
    The protocol is an `RpcGroup`, so it runs over any `RpcClient` protocol, through a relay or a server that decides what's accepted.

    ```ts
    yield* Relay.connect(notes, options)
    // In one process: Relay.layer
    // Over a network: Relay.layerRemote
    ```
